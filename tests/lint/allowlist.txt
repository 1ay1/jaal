# Files allowed to use raw concurrency primitives, and which ones.
# Each is jaal's own implementation of a safe type: the raw primitive is
# what the safe type is built from. Anything not listed here must go
# through the safe types. See tests/lint/banlist.cmake.

# Sink is defined here; sink_access is its private minting key.
core/sink.hpp: sink_access
# Task map builds a forwarding sink for the mapped task.
core/fx.hpp: sink_access
# Stream map builds a forwarding sink for the mapped stream, the same way.
core/stream.hpp: sink_access
# The kernel mints each stream run's generation-gated sink. `atomic`: a
# lock-free "a task failed" flag, so an idle step doesn't take a lock.
kernel/kernel.hpp: sink_access atomic
# given<P> runs tasks inline and mints a sink into a local vector for them.
host/given.hpp: sink_access

# The mailbox IS the lock around the message queue; it mints sinks to itself.
# `thread`: it records std::thread::id of the loop (no thread is created)
# so a full blocking mailbox refuses a loop-thread post instead of
# deadlocking on a drain only the loop can do.
# `atomic`: a lock-free "maybe non-empty" hint so an idle drain skips the lock.
kernel/mailbox.hpp: mutex sink_access thread atomic
# loop_key is defined here.
kernel/loop.hpp: loop_key
# The worker pool: threads, the job queue lock, and detached isolated threads.
kernel/pool.hpp: thread detach mutex
# scope: joins every helper thread; `observed` is an atomic flag per helper.
kernel/scope.hpp: thread atomic
# The "who waits on this thread" set that lets a mailbox refuse a post
# that would deadlock (a scope helper sending to the loop that joins it).
kernel/waits.hpp: thread thread_local
# guarded<T>: the shared_mutex it wraps.
kernel/guarded.hpp: mutex

# The signal handler's test hook is an atomic (the only thing a handler
# may touch). Declared here so tests and the .cpp agree on it.
platform/posix/signals.hpp: atomic
